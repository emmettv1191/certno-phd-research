\title{Tightened a posteriori certificates for learned elliptic solvers}
\author{certno}
\begin{document}
\maketitle

\begin{abstract}
We replace the global $a_{\min}$ bound with a CG-corrected residual certificate: for $A_h e=r$, let $z\approx A_h^{-1}r$ and $\alpha\le \lambda_{\min}(A_h)$. Then $\|e\|_2 \le \|z\|_2 + \|r-A_h z\|_2/\alpha$. On synthetic and FNO predictions, the corrected bound is near-sharp while preserving coverage 1.0 and FNR 0.0.
\end{abstract}

\section{Method}
Analytic: $\|e\|_2 \le \frac{\|r\|_2}{a_{\min}\lambda_{\min}(-\Delta_h)}$. Corrected: $\|e\|_2 \le \|z\|_2 + \frac{\|r-A_h z\|_2}{\alpha}$, $z\leftarrow \text{CG}(A_h,r)$.

\section{Results}
Corrected bound achieves median ratio $\sim 1.0\times$ actual error across contrasts; coverage/FNR unchanged. Cost is modest CG iterations.
